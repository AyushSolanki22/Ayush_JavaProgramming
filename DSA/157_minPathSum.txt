//One approach : recursion + Memorisation

class Solution {
    static int dp[][];
    public int minPathSum(int[][] grid) {
        dp=new int[grid.length][grid[0].length];
        for(int row[]: dp){
            Arrays.fill(row, 0);
        }
        return pathSum(grid, 0, 0);
    }

    public int pathSum(int[][] grid, int i, int j) {

        int m = grid.length, n = grid[0].length;
        // BASE CASE 1:
        // If we move outside the grid, this is an invalid path.
        if (i >= m || j >= n)  return Integer.MAX_VALUE;

        // BASE CASE 2:
        // If we reach the destination, return its cell value.
        if (i == m - 1 && j == n - 1) return grid[i][j];

        if(dp[i][j]!=0) return dp[i][j];

        // CHOICE 1: Move DOWN
        int down = pathSum(grid, i + 1, j);

        // CHOICE 2: Move RIGHT
        int right = pathSum(grid, i, j + 1);

        // Choose the path with minimum cost.
        return dp[i][j]=grid[i][j] + Math.min(down, right);
    }
}


// PROBLEM: Minimum Path Sum
// APPROACH: Recursion + Memoization (Top-Down DP)

// GOAL:
// Find the minimum sum of values along a path
// from the top-left cell (0,0) to the bottom-right cell.
// We can only move RIGHT or DOWN.

// 1. DP STATE:
// dp[i][j] stores the minimum path sum required
// to travel from the current cell (i,j) to the destination
// (m-1,n-1).

// 2. BASE CASES:

// CASE 1: OUTSIDE THE GRID
// If i >= m or j >= n, we have moved outside the grid.
// This is an invalid path, so return Integer.MAX_VALUE.

// CASE 2: DESTINATION REACHED
// If i == m-1 and j == n-1,
// return grid[i][j] because we have reached the destination.

// 3. MEMOIZATION CHECK:
// If dp[i][j] is already calculated,
// return its stored value instead of calculating it again.
// This avoids repeated recursive calls.

// 4. CHOICES:

// CHOICE 1: MOVE DOWN
// Move to the next row while staying in the same column.
// down = pathSum(i+1, j)

// CHOICE 2: MOVE RIGHT
// Move to the next column while staying in the same row.
// right = pathSum(i, j+1)

// 5. RECURRENCE:
// Add the current cell's value to the minimum
// of the two possible paths.
//
// dp[i][j] = grid[i][j] + min(down, right)

// 6. STORE THE RESULT:
// Save the calculated minimum path sum in dp[i][j].
// This ensures that the same state is never recalculated.

// 7. FINAL ANSWER:
// Start from the top-left cell.
// pathSum(0,0)

// TIME COMPLEXITY: O(m*n)
// SPACE COMPLEXITY: O(m*n) for DP + O(m+n) recursion stack






//Another Approach :  tabulation method


//create a dp array of m*n size  ((same size as that of array))

//intialisation : dp[i][j] will contain the minSum to reach to there from start (0, 0)

//filling first row, second row is very simple ,,,, 
//filling middle elements will be grid[i][j]+ Math.min(left, upper) elements

class Solution {
    public int minPathSum(int[][] grid) {
        int m=grid.length, n=grid[0].length;    

        int dp[][]=new int[m][n]; 

        dp[0][0]=grid[0][0]; 
        //filling first col
        for(int i=1; i<m; ++i) dp[i][0]=grid[i][0]+dp[i-1][0]; 
        //filling first row
        for(int j=1; j<n; ++j) dp[0][j]+=grid[0][j]+dp[0][j-1]; 

        //filling middle elements
        for(int i=1; i<m; ++i){
            for(int j=1; j<n; ++j){
                dp[i][j]=grid[i][j]+Math.min(dp[i-1][j], dp[i][j-1]); 
            }
        }

        return dp[m-1][n-1];
    }
}




// PROBLEM: Minimum Path Sum
// APPROACH: Bottom-Up DP (Tabulation)

// GOAL:
// Find the minimum sum of values along a path
// from the top-left cell (0,0) to bottom-right cell (m-1,n-1).
// We can only move RIGHT or DOWN.

// 1. DP STATE:
// Create a 2D DP array of size m*n.
//
// dp[i][j] stores the minimum path sum required
// to reach cell (i,j) FROM the starting cell (0,0).

// 2. INITIALIZATION:
// STARTING CELL:
// dp[0][0] = grid[0][0]
// Because we are already standing at the starting cell.

// 3. FIRST COLUMN:
// Every cell in the first column can only be reached
// by moving DOWN.
// Therefore, add the current cell value to the previous cell.
//
// dp[i][0] = grid[i][0] + dp[i-1][0]

// 4. FIRST ROW:
// Every cell in the first row can only be reached
// by moving RIGHT.
// Therefore, add the current cell value to the previous cell.
//
// dp[0][j] = grid[0][j] + dp[0][j-1]

// 5. FILLING THE REMAINING CELLS:
// For every middle cell (i,j), we have two ways to reach it:
//
// CHOICE 1: COME FROM TOP
// We reach the current cell by moving DOWN from (i-1,j).
// Minimum cost = dp[i-1][j]
//
// CHOICE 2: COME FROM LEFT
// We reach the current cell by moving RIGHT from (i,j-1).
// Minimum cost = dp[i][j-1]
//
// Choose the smaller cost and add the current cell's value.
//
// dp[i][j] = grid[i][j] + min(dp[i-1][j], dp[i][j-1])

// 6. FINAL ANSWER:
// dp[m-1][n-1] stores the minimum path sum
// from the starting cell to the destination.

// TIME COMPLEXITY: O(m*n)
// SPACE COMPLEXITY: O(m*n)
