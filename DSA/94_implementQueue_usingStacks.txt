class MyQueue {
    Stack<Integer> st=new Stack<>();  //main  stack

    public MyQueue() {
        
    }
    
    public void push(int x) {
        st.push(x);
    }
    
    public int pop() {
        Stack<Integer> helper=new Stack<>();   //helper stack 
        while(st.size()>1){
            helper.add(st.pop());
        }
        int popVal=st.pop();
        while(helper.size()>0){
            st.push(helper.pop());
        }
        return popVal;
    }
    
    public int peek() {
        Stack<Integer> helper=new Stack<>();   //helper stack 
        while(st.size()>1){
            helper.add(st.pop());
        }
        int popVal=st.peek();
        while(helper.size()>0){
            st.push(helper.pop());
        }
        return popVal;
    }
    
    public boolean empty() {
        return (st.size()==0);
    }
}
