//One Approach -- storing bst inorder in to array, and thus getting kth smallest, returning arr[k-1]  ((as inorder is sorted))

//T.C = 0(n), A.S= 0(n)



//Second Approach -- using global variable k, ans to track of inorder and thus mark kth value to the ans, and thus return ans;


class Solution {
    int k;
    int ans;

    public int kthSmallest(TreeNode root, int k) {
        this.k = k;   //storing k value's in global variable
        inorder(root);
        return ans;
    }

    public void inorder(TreeNode root) {
        if(root == null) return;

        // Go to smaller values first
        inorder(root.left);

        // Visit current node
        k--;

        if(k == 0) {
            ans = root.val;
            return;
        }

        // Then go to larger values
        inorder(root.right);
    }
}
