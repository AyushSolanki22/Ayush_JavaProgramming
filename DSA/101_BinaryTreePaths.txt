// Maintain the current root-to-node path as a String.
//
// At every node:
// 1. Append the current node value to the path.
// 2. If it is a leaf node, the current path becomes a
//    complete root-to-leaf path, so add it to the answer.
// 3. Recursively explore the left and right subtrees.
//
// No backtracking is required because String is immutable.
// Every recursive call gets its own copy of the path.


class Solution {
    public List<String> binaryTreePaths(TreeNode root) {
        List <String> ans=new ArrayList<>();
        String path="";

        traversal(root,ans,path);
        return ans;
    }
    
    public void traversal(TreeNode root, List<String> ans, String path){
        if(root==null) return;

        if(path=="") path+=Integer.toString(root.val);
        else path+="->"+Integer.toString(root.val);

        if(root.left==null && root.right==null){
            ans.add(path);
            path="";
        }

        traversal(root.left, ans, path);
        traversal(root.right, ans, path);



    }
}
