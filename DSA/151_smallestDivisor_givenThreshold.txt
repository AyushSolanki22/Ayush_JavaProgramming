class Solution {
    public int smallestDivisor(int[] nums, int threshold) {
        //binary search on answer   //binary search on smallest divisor
        
        int start=1, end=nums[0] ;
        for(int ele:nums){
            end=Math.max(end, ele);
        }
        
        int ans=end;
        
        while(start<=end){
            int mid=start+(end-start)/2; 
            
            int sum=sum(nums, mid);
            
            if(sum<=threshold){    //ans may be found, but search for better answer
                ans=mid; 
                end=mid-1;  
            }
            else {    //divisor not valid, need of larger divisor
                start=mid+1;
            }
        }
        
        return ans;
    }

    int sum(int nums[], int divisor){
        int sum=0; 
        for(int ele : nums){
            sum+=Math.ceil((double)ele/divisor);   //double typecasting as we need decimal division
        }
        ;
        return sum;
    }
}
