// (using for loop)

class Solution {
    public int minBitFlips(int start, int goal) {
        int temp=start^goal;       //count of 1 sets using ex-or -- gives how many digits need flip ((how many digits of start are different from goal))

        //counting sets 
        int count=0; 
        for(int i=0; i<31; ++i){
            if((temp>>i)%2==1) count++;
        }

        return count;
    }
}



// Time Complexity:
//
// The for loop executes exactly 31 times.
// Each iteration performs constant-time operations.
// Since the number of iterations is fixed and does not
// depend on the input values, time complexity is O(1).
//
// Space Complexity: O(1)
// Only a fixed number of variables are used.
