class Solution {
    public int rob(int[] nums) {

        int n = nums.length;

        // Edge case: Only one house exists.
        if (n == 1) return nums[0];

        // CASE 1: Exclude the last house.
        int firstHouse = robI(nums, 0, n - 2);

        // CASE 2: Exclude the first house.
        int lastHouse = robI(nums, 1, n - 1);

        // Return the maximum money from both cases.
        return Math.max(firstHouse, lastHouse);
    }

    // Normal House Robber logic for a given range.
    public int robI(int[] arr, int start, int end) {

        int n = end - start + 1;

        int[] dp = new int[n];

        // Base cases
        dp[0] = arr[start];

        if (n > 1) {
            dp[1] = Math.max(arr[start], arr[start + 1]);
        }

        // Apply the normal House Robber recurrence.
        for (int i = 2; i < n; i++) {

            // Choice 1: Pick current house.
            int pick = arr[start + i] + dp[i - 2];

            // Choice 2: Skip current house.
            int skip = dp[i - 1];

            dp[i] = Math.max(pick, skip);
        }

        return dp[n - 1];
    }
}




// PROBLEM: House Robber II (LeetCode 213)

// GOAL:
// Houses are arranged in a circle.
// We cannot rob two adjacent houses.
// The first and last houses are also adjacent.
// Find the maximum amount of money we can rob.

// 1. KEY OBSERVATION:
// Since the first and last houses are adjacent,
// we cannot rob both of them.

// Therefore, we have two possible cases:

// CASE 1: DON'T ROB THE LAST HOUSE
// Consider houses from index 0 to n-2.
// We are free to rob the first house.

// CASE 2: DON'T ROB THE FIRST HOUSE
// Consider houses from index 1 to n-1.
// We are free to rob the last house.

// 2. SOLVE EACH CASE:
// Both cases become the original House Robber (LeetCode 198).

// For each house, we have two choices:

// CHOICE 1: PICK CURRENT HOUSE
// Rob the current house and skip the previous one.
// pick = nums[i] + dp[i-2]

// CHOICE 2: SKIP CURRENT HOUSE
// Don't rob the current house.
// skip = dp[i-1]

// 3. RECURRENCE:
// dp[i] = max(pick, skip)

// 4. FINAL ANSWER:
// Take the maximum result from both cases:
//
// max(rob houses 0 to n-2,
//     rob houses 1 to n-1)

// 5. WHY DOES THIS WORK?
// Every valid solution must exclude either the first house
// or the last house (or both).
// By solving both possibilities, we cover every valid solution.

// TIME COMPLEXITY: O(n)
// SPACE COMPLEXITY: O(n) with DP array
