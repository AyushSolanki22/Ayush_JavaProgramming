/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */

class Solution {
    public boolean isSymmetric(TreeNode root) {
        if(root==null) return true;
        
        return isMirror(root.left,root.right);
    }
    
    public Boolean isMirror(TreeNode left, TreeNode right){
        if(left==null && right==null) return true;
        
        if(left==null || right==null) return false;
        if(left.val!=right.val) return false;
        
        return isMirror(left.left, right.right) && isMirror(left.right,right.left);
    }
    
}



// Symmetric means left subtree should be a mirror of right subtree.
//
// Compare two trees simultaneously.
//
// 1. Both null -> symmetric.
// 2. One null -> not symmetric.
// 3. Values differ -> not symmetric.
//
// Otherwise,
// compare:
// left.left with right.right
// left.right with right.left
//
// (Cross comparison because we are checking mirror images.)
