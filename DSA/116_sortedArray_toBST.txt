class Solution {
    public TreeNode sortedArrayToBST(int[] nums) {
        return build(nums, 0, nums.length-1 );
    }

    public TreeNode build(int nums[], int lo, int hi){
        if(lo>hi) return null;

        int mid=lo+(hi-lo)/2;
        TreeNode root=new TreeNode(nums[mid]);

        root.left=build(nums, lo, mid-1);
        root.right=build(nums, mid+1, hi);
        return root; 
    }
}









//INTUITION

// SORTED ARRAY TO BALANCED BST
//
// A BST is height-balanced when we choose the middle element
// as the root.
//
// Why?
//
// All elements smaller than nums[mid] will go to the left subtree.
// All elements greater than nums[mid] will go to the right subtree.
//
// Since the array is already sorted:
//
//     left half  = smaller elements
//     middle     = root
//     right half = greater elements
//
// Therefore:
//
//     root.left  = build(left half)
//     root.right = build(right half)
//
// We repeat the same process for every half.
//
// ------------------------------------------------------------
//
// EXAMPLE:
//
// nums = [-10, -3, 0, 5, 9]
//
// Middle element:
//
//     mid = 2
//     nums[mid] = 0
//
// So:
//
//          0
//         / \
//       -10  5
//         \   \
//         -3   9
//
// More accurately, the recursive construction is:
//
//          0
//         / \
//       -10   5
//         \     \
//         -3     9
//
// ------------------------------------------------------------
//
// STEP 1: FIND MIDDLE
//
//     int mid = lo + (hi - lo) / 2;
//
// We use this formula instead of:
//
//     (lo + hi) / 2
//
// because lo + hi can overflow for very large indexes.
//
// The middle element becomes the root of the current subtree.
//
// ------------------------------------------------------------
//
// STEP 2: CREATE ROOT
//
//     TreeNode root = new TreeNode(nums[mid]);
//
// The current middle value is selected as the root.
//
// Since the array is sorted:
//
//     nums[lo ... mid-1]
//
// contains values smaller than the root.
//
// And:
//
//     nums[mid+1 ... hi]
//
// contains values greater than the root.
//
// ------------------------------------------------------------
//
// STEP 3: BUILD LEFT SUBTREE
//
//     root.left = build(nums, lo, mid - 1);
//
// The left subtree uses the left half of the current range.
//
// Important:
//
//     Start index = lo
//     End index   = mid - 1
//
// We must use lo, not 0,
// because the current recursive call may already be working
// on a smaller part of the array.
//
// ------------------------------------------------------------
//
// STEP 4: BUILD RIGHT SUBTREE
//
//     root.right = build(nums, mid + 1, hi);
//
// The right subtree uses the right half of the current range.
//
// Important:
//
//     Start index = mid + 1
//     End index   = hi
//
// Your code currently uses:
//
//     root.right = build(nums, mid + 1, 0);
//
// This is incorrect because the ending index should be hi,
// not 0.
//
// ------------------------------------------------------------
//
// STEP 5: BASE CASE
//
//     if(lo > hi) return null;
//
// If the current range becomes invalid,
// there are no elements left to create a node.
//
// Therefore, return null.
//
// This null is assigned to the left or right child
// of a leaf node.
//
// ------------------------------------------------------------
//
// WHY DO WE NOT NEED reverse()?
//
// Your array is already sorted.
//
// Reversing the left and right portions will destroy
// the sorted order required for constructing a BST.
//
// Example:
//
// Original:
//
//     [-10, -3, 0, 5, 9]
//
// After reversing:
//
//     [-3, -10, 0, 9, 5]
//
// This is no longer sorted.
//
// Therefore, remove both lines:
//
//     reverse(nums, 0, mid - 1);
//     reverse(nums, mid + 1, hi);
//
// Also, the reverse() method is not needed at all.
//
// ------------------------------------------------------------
//
// DRY RUN:
//
// nums = [-10, -3, 0, 5, 9]
//
// FIRST CALL:
//
//     build(nums, 0, 4)
//
// mid:
//
//     mid = 0 + (4 - 0) / 2
//     mid = 2
//
// root:
//
//     nums[2] = 0
//
// Create:
//
//          0
//
// Left call:
//
//     build(nums, 0, 1)
//
// Right call:
//
//     build(nums, 3, 4)
//
// ------------------------------------------------------------
//
// LEFT CALL:
//
//     build(nums, 0, 1)
//
// mid:
//
//     mid = 0 + (1 - 0) / 2
//     mid = 0
//
// root:
//
//     nums[0] = -10
//
// Create:
//
//        -10
//
// Left call:
//
//     build(nums, 0, -1)
//
// Since:
//
//     lo > hi
//
// Return null.
//
// Right call:
//
//     build(nums, 1, 1)
//
// mid:
//
//     mid = 1
//
// root:
//
//     nums[1] = -3
//
// Therefore:
//
//        -10
//           \
//           -3
//
// ------------------------------------------------------------
//
// RIGHT CALL:
//
//     build(nums, 3, 4)
//
// mid:
//
//     mid = 3 + (4 - 3) / 2
//     mid = 3
//
// root:
//
//     nums[3] = 5
//
// Left call:
//
//     build(nums, 3, 2)
//
// Return null.
//
// Right call:
//
//     build(nums, 4, 4)
//
// root:
//
//     nums[4] = 9
//
// Therefore:
//
//        5
//         \
//          9
//
// ------------------------------------------------------------
//
// FINAL TREE:
//
//          0
//         / \
//       -10   5
//         \     \
//         -3     9
//
// This is a valid height-balanced BST.
//

// TIME COMPLEXITY:
//
// Every array element is used exactly once
// to create one tree node.
//
// Therefore:
//
//     Time Complexity = O(n)
//
// Each recursive call divides the array range into halves.
//
// Recursion stack height:
//
//     O(log n)
//
// Therefore:
//
//     Auxiliary Space = O(log n)
//
// The output tree itself contains n nodes:
//
//     Output Space = O(n)
//
// ------------------------------------------------------------
//
// ONE-LINE INTUITION:
//
// Choose the middle element as root,
// recursively build the left half as the left subtree,
// and recursively build the right half as the right subtree.
