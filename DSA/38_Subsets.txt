//One Approach  (using Recursion)

class Solution {

    public List<List<Integer>> subsets(int[] nums) {

        List<List<Integer>> finalAns = new ArrayList<>();

        List<Integer> subset = new ArrayList<>();

        helper(nums, 0, subset, finalAns);

        return finalAns;
    }

    static void helper(int nums[],
                       int idx,
                       List<Integer> subs,
                       List<List<Integer>> finalAns){

        // base case
        if(idx == nums.length){

            finalAns.add(new ArrayList<>(subs));   //new arraylist here to add a copy, not a original reference of arrayList , otherwise all subs added in final ans will be poiting to same subs
            //For mutable objects like:  ArrayList, StringBuilder, arrays,,, storing variable stores: reference/address

            return;
        }

        // skip
        helper(nums, idx+1, subs, finalAns);

        // take
        subs.add(nums[idx]);

        helper(nums, idx+1, subs, finalAns);

        // backtrack
        subs.remove(subs.size()-1);
    }
}




//Another approach -- (using Bit Manipulation)


class Solution {
    public List<List<Integer>> subsets(int[] nums) {
        int n=nums.length; 
        int m=1<<n;   //2^n

        List<List<Integer>> ans=new ArrayList<>();
        for(int i=0; i<m ; ++i){     //no. pe map krna -- yee numbers ki set digits subsets denote krte hai for elements 
            List<Integer> subset=new ArrayList<>();

            for(int j=0; j<n; ++j){
                if((i>>j)%2==1) subset.add(nums[j]);
            }
            ans.add(subset);
        }

        return ans;
    }
}



// Subsets using bit masks
//
// 1. For n array elements, there are 2^n possible subsets.
// 2. Run i from 0 to (2^n - 1).
// 3. Treat i as a binary mask with n bits.
// 4. Bit position j corresponds to arr[j].
// 5. If bit j is 1, include arr[j] in the subset.
// 6. If bit j is 0, exclude arr[j] from the subset.
// 7. Store each generated subset in the answer.
//
// Example: arr = {3, 7, 4}, i = 5
// i = 5 = binary 101
// Bit 0 = 1 -> include 3
// Bit 1 = 0 -> exclude 7
// Bit 2 = 1 -> include 4
// Generated subset = {3, 4}
//
// Number of subsets: 2^n
// Time: O(n * 2^n)
// Space: O(n * 2^n) to store all subsets


i = 0  -> 000 -> {}
i = 1  -> 001 -> {3}
i = 2  -> 010 -> {7}
i = 3  -> 011 -> {3, 7}
i = 4  -> 100 -> {4}
i = 5  -> 101 -> {3, 4}
i = 6  -> 110 -> {7, 4}
i = 7  -> 111 -> {3, 7, 4}
