//Hint: a^a=0 ,,,, 0^b=b 

//we ex-or all the numbers in a loop, and thus get the ans ---- single occuring element remains 

class Solution {
    public int singleNumber(int[] nums) {
        int ans=0;

        for(int i=0; i<nums.length; ++i){
            ans=ans^nums[i];
        }

        return ans;
    }
}
