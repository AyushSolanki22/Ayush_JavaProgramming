//One Solution -- using queue     //T.C= O(nk)  S.C=O(n)
//concept -- we will use queue as it can work as circular data , helpful in this ques

class Solution {
    public int findTheWinner(int n, int k) {
            Queue <Integer> q=new LinkedList<>();
            
            for(int i=1; i<=n; ++i){
                q.add(i);
            }

            while(q.size()>1){
                for(int i=1; i<=k-1; ++i){
                    q.add(q.remove());
                }
                q.remove();
            }

            return q.peek();
    }
}



//Another Solution    T.C= O(n)  S.C=O(1)    //either recursive or iterative 
//Josephus recurrence
//Concept
//f(1) = 0

//f(n) = (f(n-1) + k) % n

The winner is: f(n)+1


class Solution {

    public int josephus(int n, int k) {
        if (n == 1)
            return 0;

        return (josephus(n - 1, k) + k) % n;
    }

    public int findTheWinner(int n, int k) {
        return josephus(n, k) + 1; // Convert to 1-based indexing
    }
}
