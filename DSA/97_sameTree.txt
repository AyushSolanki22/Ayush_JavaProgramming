//One approach (using static variable)

/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */

class Solution {
    boolean ans=true;
    public boolean isSameTree(TreeNode p, TreeNode q) {
        traversal(p,q);
        return ans;
    }
    public void traversal(TreeNode p, TreeNode q){
        if(p==null && q==null) return; 
        else if (p==null || q==null) {
            ans=false;   return;
        }

        if(p.val!=q.val){
            ans=false; return; 
        }

        traversal(p.left, q.left);
        traversal(p.right, q.right);
        
    }
}




//another approach 
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */

class Solution {
    public boolean isSameTree(TreeNode p, TreeNode q) {
        if(p==null && q==null) return true;
        else if (p==null || q==null) return false;

        if(p.val!=q.val)  return false;

        return isSameTree(p.left, q.left) && isSameTree(p.right, q.right);

    }
}



//Intuition

// Compare both trees simultaneously.
//
// Base Cases:
// 1. Both nodes are null -> trees are same.
// 2. One node is null -> trees differ.
// 3. Node values differ -> trees differ.
//
// Otherwise, recursively compare
// left subtree and right subtree.
// Both must be identical.
