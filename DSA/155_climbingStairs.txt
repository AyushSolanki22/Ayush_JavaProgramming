//One approach : using recursion + Memorisation

class Solution {
    static int dp[]; 
    public int climbStairs(int n) {
        dp = new int[n + 1];
        Arrays.fill(dp, -1);

        return count(0, n);
    }

    public int count(int i, int n) {   //count ways
        if (i==n) return 1;
        if (i>n) return 0;

        if (dp[i]!=-1) return dp[i];

        int oneStep = count(i+1, n);
        int twoStep = count(i+2, n);

        return dp[i] = oneStep+twoStep;
    }
}



// Goal: Count the number of distinct ways to reach stair n.

// At every stair i, we have 2 choices:

// 1. Take 1 step -> count(i+1)
// 2. Take 2 steps -> count(i+2)

// Total ways = ways from both choices.

// Recurrence:
// count(i) = count(i+1) + count(i+2)

// Base cases:

// If i == n:
// We successfully reached the destination.
// Return 1 because we found ONE valid way.

// If i > n:
// We crossed the destination.
// Return 0 because this is NOT a valid way.

// Important:
// We return 1 at the destination, not 0.
// Each successful recursive path contributes exactly 1 way.





//Another Approach : using tabulation


class Solution {
    public int climbStairs(int n) {
        int dp[]=new int[n+1];

        for(int i=0; i<=n; ++i){
            if(i<=2) dp[i]=i;
            else dp[i]=dp[i-1]+dp[i-2];
        }

        return dp[n];
    }
}



// PROBLEM: Climbing Stairs (LeetCode 70)
// APPROACH: Bottom-Up DP (Tabulation)

// GOAL:
// Find the total number of distinct ways to reach the nth stair.
// At each step, we can climb either 1 stair or 2 stairs.

// 1. DP STATE:
// Create a 1D DP array of size n+1.
//
// dp[i] stores the total number of distinct ways
// to reach the ith stair from the ground (stair 0).

// 2. BASE CASES:

// dp[0] = 1
// There is exactly one way to stay at the ground:
// take zero steps.

// dp[1] = 1
// There is only one way to reach stair 1:
// take one 1-step.

// dp[2] = 2
// There are two ways to reach stair 2:
// 1. Take two 1-steps (1 + 1).
// 2. Take one 2-step (2).

// 3. TRANSITIONS:

// To reach stair i, we have two choices:

// CHOICE 1: TAKE 1 STEP
// Come from stair (i-1).
// Number of ways = dp[i-1]

// CHOICE 2: TAKE 2 STEPS
// Come from stair (i-2).
// Number of ways = dp[i-2]

// 4. RECURRENCE:
// Add both possibilities because they represent
// two different ways of reaching stair i.
//
// dp[i] = dp[i-1] + dp[i-2]

// 5. FINAL ANSWER:
// dp[n] stores the total number of ways to reach stair n.

// TIME COMPLEXITY: O(n)
// SPACE COMPLEXITY: O(n)
