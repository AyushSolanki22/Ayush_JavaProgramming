//HARDEST QUESTION
//step 1 -- creating deep copy without random connections  (of like simple ll) -- with help of dummy node   --- h2 corresponding to deep copied list
//step 2 -- alternatively connect both LL  dummy node , (h1, h2) & t1,t2
//step 3 -- assigning random pointers - get t1,t2 to h1,h2---in loop, t2.random=t1.random.next; t1,t2=t1.next.next ,t2.next.next ,,, random==null condition -- magic of because of alternate conncetions
//step 4 -- splitting the combined list  -- create d1, d2 (dummy nodes), t1, t2 & t (for traversing combined list)

class Solution {
    public Node copyRandomList(Node head1) {
        //step-1 : Creating deep copy without random
        Node head2=deepCopy(head1);

        //step-2 : Alternatively connecting lists  --- interweave concept 
        merge(head1,head2);

        //step 3 : Assigning random pointers
        randomConnections(head1, head2);

        //step 4 : Split the list
        split(head1);

        return head2;
    }

    public Node deepCopy(Node head1){
        Node temp1=head1;   //for traversing through original list 

        Node head2=new Node(-1);  //dummy for head2
        Node ans=head2;
        while(temp1!=null){
            Node t=new Node(temp1.val);
            ans.next=t;

            ans=ans.next;
            temp1=temp1.next;
        }
        ans.next=null;
        return head2.next;
    }

    public void merge(Node head1, Node head2){
        Node t1=head1, t2=head2;
        Node dummy=new Node(-1);
        Node t=dummy;
        while(t1!=null){
            t.next=t1;  t1=t1.next;
            t=t.next;
            t.next=t2;  t2=t2.next;
            t=t.next;
        }
    }

    public void randomConnections(Node head1, Node head2){
        Node t1=head1;
        Node t2=head2;
        while(t1!=null){
            if(t1.random==null) t2.random=null;
            else t2.random=t1.random.next;

            t1=t1.next.next;
            if(t2.next!=null) t2=t2.next.next;   //if condition -- edge case
        }
    }

    public void split(Node head1){
        Node dummy1=new Node(-1);
        Node t1=dummy1;
        Node dummy2=new Node(-1);
        Node t2=dummy2;
        Node temp=head1;  //for traversing merged list
        
        while(temp!=null){
            t1.next=temp;  t1=t1.next;
            temp=temp.next;

            t2.next=temp;  t2=t2.next;
            temp=temp.next;
        }
        t1.next=null;   //to break connection at last too of merged list
    }
}





//Another Approach --- using HashMap

//original list - a
//create a copied list - b (same step 1 as of ll approach)
//simply, store deep copy (node(a), node(b)) in hashmap 

//assume a variabe ta at start of a ,,  in a loop, map.get(ta).random=map.get(ta.random);       ((b.random=a.random which is in b))


class Solution {
    public Node copyRandomList(Node head) {
        Node a=head;
        Node b=deepCopy(a);

        HashMap<Node, Node> map=new HashMap<>();
        Node t1=a, t2=b; 
        while(t1!=null && t2!=null){  //storing deepcopy in hashmap
            map.put(t1, t2); 
            t1=t1.next; 
            t2=t2.next; 
        }

        t1=a;
        while(t1!=null){
            map.get(t1).random=map.get(t1.random); 
            t1=t1.next;
        }

        return b;
    }


    public Node deepCopy(Node head1){
        Node temp1=head1;
        Node head2=new Node(-1);  //dummy for head2
        Node ans=head2;
        while(temp1!=null){
            Node t=new Node(temp1.val);
            ans.next=t;

            ans=ans.next;
            temp1=temp1.next;
        }
        ans.next=null;
        return head2.next;
    }
}






// Copy Random List:
//
// Problem:
// Every node has:
// 1. next pointer
// 2. random pointer
//
// We cannot simply copy random pointer because the random pointer
// of the copied node must point to a COPIED node, not an original node.
//
// Step 1:
// Deep copy the normal next structure.
//
// Original:
// 1 → 2 → 3
//
// Copy:
// 1' → 2' → 3'
//
// At this point:
// next = copied
// random = not copied
//
// Step 2:
// Create HashMap:
//
// Original Node → Copied Node
//
// 1 → 1'
// 2 → 2'
// 3 → 3'
//
// HashMap acts as a translator from original nodes to their copies.
//
// Step 3:
// Copy random pointers.
//
// Suppose:
// 1.random → 3
// 
// We need:
// 1'.random → 3'
//
// map.get(1) gives 1'
// map.get(1.random) gives 3'
//
// Therefore:
//
// map.get(t1).random = map.get(t1.random);
//
// If t1.random == null:
// map.get(null) gives null.
//
// Main idea:
// Whenever we have an ORIGINAL node and need its COPIED version,
// use:
//
// map.get(originalNode)
//
// Time Complexity: O(n)
// Space Complexity: O(n)
