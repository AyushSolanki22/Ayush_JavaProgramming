//Using maxHeap

class Solution {
    public static int kthLargest(int arr[], int k) {
        // code here
        PriorityQueue<Integer> p=new PriorityQueue<>();
        
        for(int i=0; i<arr.length; ++i){
            p.add(arr[i]);
            if(p.size()>k) p.remove();
        }
        
       
        return p.peek();
    }
}
