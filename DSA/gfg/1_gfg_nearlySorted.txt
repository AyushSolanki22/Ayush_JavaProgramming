class Solution {
    public void nearlySorted(int[] arr, int k) {
        // the elements in this nearly sorted array, has been shifted max to max k positions away only , from the original supposed to be location in sorted array
        
        PriorityQueue<Integer> p=new PriorityQueue<>();
        
        int pos=0;
        for(int i=0; i<arr.length; ++i){
            p.add(arr[i]);
            if(p.size()>k) arr[pos++]=p.remove();
        }
        
        while(p.size()>0) arr[pos++]=p.remove();
        
    }
}
