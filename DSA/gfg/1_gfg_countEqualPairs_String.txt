// My self approach 

class Solution {
    public int equalPairs(String s) {
        HashMap<Character, Integer> map=new HashMap<>();
        
        for(int i=0; i<s.length(); ++i){
            char ch=s.charAt(i);
            map.put(ch, map.getOrDefault(ch,0)+1); 
            
        }
        
        int count=0;
        for(char x: map.keySet()){
            if(map.get(x)>1) count+=map.get(x)*(map.get(x)-1);
        }  //count gives all linked pairs to each other for with having freq>1 as they will be only having external pairs(excluding the i==j pairs ---eg- "aa"-2 , "aaa"-6)
        
        return count+s.length();   //s.length gives all pairs with i==j
        
    }
}


// Another Approach

class Solution {
    public int equalPairs(String s) {
        //ananya -- 3+2+3+2+1+3

        HashMap<Character, Integer> map=new HashMap<>();
        
        for(int i=0; i<s.length(); ++i){
            char ch=s.charAt(i);
            map.put(ch, map.getOrDefault(ch,0)+1); 
        }
        
        int pairs=0;
        for(char x: map.keySet()){
            int freq=map.get(x);
            pairs+=freq*freq;
        }
        
        return pairs;
        
    }
}
