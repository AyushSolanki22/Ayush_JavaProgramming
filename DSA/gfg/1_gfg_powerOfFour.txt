class Solution {
    boolean isPowerOfFour(int n) { //it should be power of 2...and should also be perfet square
        
        return isPowerOfTwo(n) && isSquare(n);
        
    }
    
    
    public boolean isPowerOfTwo(int n) {    
        return n>0 && (n&(n-1))==0;
    }
    
    public boolean isSquare(int n) { 
        int root=(int)(Math.sqrt(n));
        return (root*root==n);
    }
}
