class Solution {
    int replaceBit(int n, int k) {
        int totalBits = 32 - Integer.numberOfLeadingZeros(n);

        if (k > totalBits) return n;

        int position = totalBits - k;
        return n & ~(1 << position);
    }
    
}


// Replace the kth bit (1-based, counted from the left) with 0.
//
// 1. Find the number of significant bits in n.
// 2. If k exceeds the number of bits, return n unchanged.
// 3. Convert the left-based position to a right-based index:
//    position = totalBits - k
// 4. Create a mask with 1 at that position using (1 << position).
// 5. Invert the mask so the target bit is 0 and all others are 1.
// 6. AND with n to clear only the target bit.
//
// Example: n = 13 (1101), k = 2
// totalBits = 4
// position = 4 - 2 = 2
//
//   1101
// & 1011
//   ----
//   1001 = 9
//
// Time: O(1)
// Space: O(1)
