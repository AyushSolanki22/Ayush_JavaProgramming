class Solution {
    public Node preToBST(int pre[]) {
        int in[]=Arrays.copyOf(pre, pre.length);
        Arrays.sort(in);
        int n=in.length; 
        
        return build(0,n-1, 0, n-1, in, pre); 
        
    }
    
    public static Node build(int preLo, int preHi, int inLo, int inHi, int inorder[], int preorder[]){
                if(preLo>preHi || inLo>inHi) return null;    //base case ****

                int val=preorder[preLo];
                Node root=new Node(val);

                int r=-1;
                for(int i=0; i<=inHi; ++i){   //locating root in inorder array
                    if(inorder[i]==val) {
                        r=i; 
                        break; 
                    }
                }

                int count=r-inLo;  //no. of elements in lst

                root.left=build(preLo+1, preLo+count, inLo, r-1, inorder, preorder); //lst call 
                root.right=build(preLo+count+1, preHi, r+1, inHi, inorder, preorder); //rst call
                //for lst, call ---- (preLo+1, preLo+count, inLo, r-1)
                //for rst, call ---- (preLo+count+1, preHi, r+1, inHi)
                return root; 
            }
}









//INTUITION


// Full intuition: Construct Binary Search Tree from Preorder Traversal
//
// Problem:
// We are given the preorder traversal of a BST.
// We have to construct the original binary tree.
//
// Example:
// preorder = [8, 5, 1, 7, 10, 12]
//
// First understand preorder:
//
// Preorder traversal follows:
// ROOT -> LEFT -> RIGHT
//
// Therefore, the first element of preorder is always the root.
//
// In this example:
// preorder[0] = 8
//
// So 8 becomes the root of the tree.
//
//                 8
//
//
// Now use the BST property:
//
// In a Binary Search Tree:
// All values smaller than the root go to the left subtree.
// All values greater than the root go to the right subtree.
//
// For root 8:
//
// Smaller values: [5, 1, 7]
// Greater values: [10, 12]
//
// Therefore:
//
//                 8
//                / \
//       [5,1,7]   [10,12]
//
//
// But we still need to know the exact structure of both subtrees.
//
// For that, we use inorder traversal.
//
// Inorder traversal follows:
// LEFT -> ROOT -> RIGHT
//
// In a BST, inorder traversal is always sorted.
//
// So if preorder is:
//
// [8, 5, 1, 7, 10, 12]
//
// We make a copy of preorder and sort it:
//
// inorder = [1, 5, 7, 8, 10, 12]
//
// Now we have:
//
// Preorder:
// ROOT -> LEFT -> RIGHT
//
// Inorder:
// LEFT -> ROOT -> RIGHT
//
// Preorder tells us which node is the root.
// Inorder tells us which nodes belong to the left and right subtree.
//
// This is the same idea as constructing a binary tree
// from preorder and inorder traversal.
//
// Important:
// We do not sort the original preorder array,
// because preorder must remain unchanged to identify roots.
//
// We create a separate copy:
//
// int[] inorder = Arrays.copyOf(preorder, preorder.length);
//
// Then sort only the copy:
//
// Arrays.sort(inorder);
//
//
// --------------------------------------------------
// Step 1: Select the root
// --------------------------------------------------
//
// The first element of the current preorder range is the root.
//
// For the complete tree:
//
// preorder = [8, 5, 1, 7, 10, 12]
//
// Root = preorder[0] = 8
//
// Create:
//
// TreeNode root = new TreeNode(8);
//
//
// --------------------------------------------------
// Step 2: Find the root in inorder
// --------------------------------------------------
//
// Inorder:
//
// [1, 5, 7, 8, 10, 12]
//           ^
//           8
//
// The index of 8 is 3.
//
// Everything before index 3 belongs to the left subtree:
//
// [1, 5, 7]
//
// Everything after index 3 belongs to the right subtree:
//
// [10, 12]
//
// Therefore:
//
// Left subtree inorder  = [1, 5, 7]
// Right subtree inorder = [10, 12]
//
//
// --------------------------------------------------
// Step 3: Calculate the size of the left subtree
// --------------------------------------------------
//
// Suppose:
//
// r = index of root in inorder
// inLo = starting index of current inorder range
//
// Then:
//
// count = r - inLo
//
// Here:
//
// r = 3
// inLo = 0
//
// count = 3 - 0
// count = 3
//
// This means that the left subtree contains 3 nodes.
//
// Why?
//
// Because inorder is:
//
// LEFT -> ROOT -> RIGHT
//
// So the number of elements before the root
// is exactly the number of nodes in the left subtree.
//
//
// --------------------------------------------------
// Step 4: Divide the preorder range
// --------------------------------------------------
//
// Original preorder:
//
// [8, 5, 1, 7, 10, 12]
//
// The first element 8 is the root.
//
// The left subtree contains 3 nodes:
//
// [5, 1, 7]
//
// The remaining nodes belong to the right subtree:
//
// [10, 12]
//
// Therefore:
//
// Left subtree preorder:
//
// preLo + 1 to preLo + count
//
// Right subtree preorder:
//
// preLo + count + 1 to preHi
//
// For the root 8:
//
// preLo = 0
// count = 3
// preHi = 5
//
// Left preorder range:
//
// 0 + 1 to 0 + 3
// 1 to 3
//
// preorder[1...3] = [5, 1, 7]
//
// Right preorder range:
//
// 0 + 3 + 1 to 5
// 4 to 5
//
// preorder[4...5] = [10, 12]
//
//
// --------------------------------------------------
// Step 5: Divide the inorder range
// --------------------------------------------------
//
// If root is found at index r:
//
// Left inorder range:
//
// inLo to r - 1
//
// Right inorder range:
//
// r + 1 to inHi
//
// The root itself is excluded from both ranges.
//
// For root 8:
//
// inorder = [1, 5, 7, 8, 10, 12]
// index of 8 = 3
//
// Left inorder range:
//
// 0 to 2
// [1, 5, 7]
//
// Right inorder range:
//
// 4 to 5
// [10, 12]
//
//
// --------------------------------------------------
// Recursive meaning of the function
// --------------------------------------------------
//
// build(preLo, preHi, inLo, inHi)
//
// means:
//
// Construct the tree using:
//
// preorder[preLo ... preHi]
//
// and:
//
// inorder[inLo ... inHi]
//
// Every recursive call constructs one complete subtree.
//
// The same process is repeated for the left and right subtree.
//
//
// --------------------------------------------------
// Base case
// --------------------------------------------------
//
// If there are no elements in the current range,
// there is no node to create.
//
// Therefore:
//
// if(preLo > preHi || inLo > inHi) {
//     return null;
// }
//
// Example:
//
// build(2, 1, 0, 0)
//
// Here preLo > preHi,
// so the range is empty.
//
// Return null.
//
//
// --------------------------------------------------
// Complete process inside build()
// --------------------------------------------------
//
// 1. Check whether the range is empty.
// 2. Take preorder[preLo] as the root.
// 3. Find the root in the inorder range.
// 4. Calculate the number of nodes in the left subtree.
// 5. Recursively build the left subtree.
// 6. Recursively build the right subtree.
// 7. Return the root.
//
//
// --------------------------------------------------
// Dry run
// --------------------------------------------------
//
// preorder = [8, 5, 1, 7, 10, 12]
// inorder  = [1, 5, 7, 8, 10, 12]
//
//
// First call:
//
// build(0, 5, 0, 5)
//
// Root:
//
// preorder[0] = 8
//
// Find 8 in inorder:
//
// r = 3
//
// Left subtree size:
//
// count = r - inLo
// count = 3 - 0
// count = 3
//
// Left subtree call:
//
// build(
//     1,
//     3,
//     0,
//     2
// )
//
// This uses:
//
// preorder[1...3] = [5, 1, 7]
// inorder[0...2]  = [1, 5, 7]
//
// Right subtree call:
//
// build(
//     4,
//     5,
//     4,
//     5
// )
//
// This uses:
//
// preorder[4...5] = [10, 12]
// inorder[4...5]  = [10, 12]
//
//
// --------------------------------------------------
// Constructing the left subtree
// --------------------------------------------------
//
// Current ranges:
//
// preorder = [5, 1, 7]
// inorder  = [1, 5, 7]
//
// First preorder element is 5.
//
// So 5 becomes the root of this subtree.
//
// In inorder:
//
// [1, 5, 7]
//     ^
//     5
//
// Values before 5:
//
// [1]
//
// Values after 5:
//
// [7]
//
// Therefore:
//
// Left child of 5 = 1
// Right child of 5 = 7
//
// Subtree:
//
//        5
//       / \
//      1   7
//
//
// --------------------------------------------------
// Constructing the right subtree
// --------------------------------------------------
//
// Current ranges:
//
// preorder = [10, 12]
// inorder  = [10, 12]
//
// First preorder element is 10.
//
// So 10 becomes the root.
//
// Inorder:
//
// [10, 12]
//   ^
//
// There are no elements before 10,
// so left subtree is null.
//
// Element 12 is after 10,
// so 12 becomes the right child.
//
// Subtree:
//
//    10
//      \
//       12
//
//
// Final tree:
//
//          8
//         / \
//        5   10
//       / \    \
//      1   7    12
//
//
// --------------------------------------------------
// Why count is necessary
// --------------------------------------------------
//
// We cannot divide preorder equally into two halves,
// because a BST does not have to be balanced.
//
// Example:
//
// preorder = [8, 5, 1, 7, 10, 12]
//
// Left subtree has 3 nodes.
// Right subtree has 2 nodes.
//
// The split is not equal.
//
// Another example:
//
// preorder = [8, 5, 1, 2, 3, 10]
//
// Most nodes may belong to the left subtree.
//
// Therefore, we must calculate the exact left subtree size
// using the root's position in inorder.
//
// count = r - inLo
//
//
// --------------------------------------------------
// Why the search starts from inLo
// --------------------------------------------------
//
// We are working with a specific inorder range:
//
// inorder[inLo ... inHi]
//
// Therefore, we should search only inside this range.
//
// Correct:
//
// int r = inLo;
//
// while(inorder[r] != rootValue) {
//     r++;
// }
//
// Incorrect:
//
// int r = 0;
//
// Starting from 0 may search outside the current subtree range.
//
// Each recursive call has its own smaller inorder range,
// so the search must begin at inLo.
//
//
// --------------------------------------------------
// Complete code
// --------------------------------------------------
//

// --------------------------------------------------
// Main intuition in one place
// --------------------------------------------------
//
// Preorder tells us the root because preorder starts with root.
//
// Since the tree is a BST, sorting the preorder values gives
// the inorder traversal.
//
// Inorder tells us how many nodes are on the left and right
// side of the current root.
//
// The number of elements before the root in inorder
// gives the size of the left subtree.
//
// Once we know the left subtree size,
// we can divide the preorder range correctly.
//
// Then we recursively repeat the same process
// for the left and right subtree.
//
// Pattern:
//
// Pick root from preorder
// Find root in inorder
// Calculate left subtree size
// Divide preorder range
// Divide inorder range
// Build left subtree
// Build right subtree
// Return root
//
//
// --------------------------------------------------
// Complexity
// --------------------------------------------------
//
// Copying preorder:
//
// O(n)
//
// Sorting inorder:
//
// O(n log n)
//
// Searching for every root linearly:
//
// O(n^2) in the worst case
//
// Therefore, total time complexity:
//
// O(n^2) worst case
//
// Extra space:
//
// O(n) for the copied inorder array
// O(h) for recursion stack
//
// Total auxiliary space:
//
// O(n + h)
//
// where:
//
// n = number of nodes
// h = height of the tree
//
//
// Important assumption:
//
// This approach assumes all values are unique.
//
// If duplicate values are present,
// preorder and inorder may not uniquely identify the tree.
