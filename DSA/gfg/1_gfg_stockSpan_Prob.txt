class Solution {
    public ArrayList<Integer> calculateSpan(int[] arr) {
        //concept
        //default case for 1st element (0th index)-- put span=1, 
        //starting from idx 1,, find prev greater element, span=idx-prevGreaterIdx  (left to right traversal )
        
        //for prev greater element, use stack while traversing, keep prev Greater index in stack - concept
        
        Stack <Integer> st=new Stack<>();
        ArrayList <Integer> span=new ArrayList<>();
        
        
        st.push(0);  span.add(1);   //st--- prevGreaterIdx ----stack mai prevGreater ka index rhega 
        
        
        for(int i=1; i<arr.length;++i){
            
            while(st.size()>0 && arr[i]>=arr[st.peek()]){
                st.pop();
            }
            
            if(!st.isEmpty())span.add(i-st.peek());
            else span.add(i+1);
            st.push(i);
        }
        
        return span;
    }
}
