class Solution {
    public int uniquePathsWithObstacles(int[][] obstacleGrid) {

        int m = obstacleGrid.length;
        int n = obstacleGrid[0].length;

        // dp[i][j] = number of paths from (0,0) to (i,j)
        int[][] dp = new int[m][n];

        // If the starting cell is blocked, no path exists.
        if (obstacleGrid[0][0] == 1) {
            return 0;
        }

        // Starting cell has exactly one path: staying there.
        dp[0][0] = 1;

        // Traverse the entire grid.
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {

                // Skip the starting cell because it is already initialized.
                if (i == 0 && j == 0) {
                    continue;
                }

                // An obstacle cannot be reached.
                if (obstacleGrid[i][j] == 1) {
                    dp[i][j] = 0;
                } 
                else {
                    // Number of ways to reach from TOP.
                    int up = (i > 0) ? dp[i - 1][j] : 0;

                    // Number of ways to reach from LEFT.
                    int left = (j > 0) ? dp[i][j - 1] : 0;

                    // Total paths = paths from TOP + paths from LEFT.
                    dp[i][j] = up + left;
                }
            }
        }

        // Return the number of paths to the destination.
        return dp[m - 1][n - 1];
    }
}



// PROBLEM: Unique Paths II
// APPROACH: Bottom-Up DP (Tabulation)

// GOAL:
// Find the number of unique paths from (0,0) to (m-1,n-1).
// We can only move RIGHT or DOWN.
// Some cells contain obstacles (1), while free cells contain (0).

// 1. DP STATE:
// Create a 2D DP array of size m*n.
//
// dp[i][j] stores the number of unique paths
// from the starting cell (0,0) to cell (i,j).

// 2. BASE CASE:
// STARTING CELL:
//
// If obstacleGrid[0][0] == 1:
//     dp[0][0] = 0;  // Starting cell is blocked., return.
//
// Otherwise:
//     dp[0][0] = 1;  // Only one way to be at the starting cell.

// 3. TRAVERSE THE GRID:
//
// For every cell (i,j):

// CASE 1: OBSTACLE
// If obstacleGrid[i][j] == 1,
// dp[i][j] = 0 because we cannot pass through an obstacle.

// CASE 2: FREE CELL
// We can reach the current cell from two directions:

// CHOICE 1: COME FROM TOP
// Number of paths = dp[i-1][j]

// CHOICE 2: COME FROM LEFT
// Number of paths = dp[i][j-1]

// TOTAL PATHS:
// dp[i][j] = dp[i-1][j] + dp[i][j-1]

// 4. FIRST ROW AND FIRST COLUMN:
// These cells have only one possible direction of entry.
// We must also ensure that no obstacle blocks the path.

// 5. FINAL ANSWER:
// Return dp[m-1][n-1].

// TIME COMPLEXITY: O(m*n)
// SPACE COMPLEXITY: O(m*n)
