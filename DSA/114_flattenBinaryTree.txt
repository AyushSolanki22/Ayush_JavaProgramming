//First Approach --- get arraylist with all nodes from traversal and make right skewed tree through arraylist

class Solution {
    public void flatten(TreeNode root) {
        ArrayList<TreeNode> nodes=new ArrayList<>();
        traversal(root, nodes);
        
        for(int i=0; i<=nodes.size()-2; ++i){
            TreeNode curr=nodes.get(i);
            curr.right=nodes.get(i+1);
        }
    }

    public static void traversal(TreeNode root, ArrayList<TreeNode> nodes) {
        if(root==null) return; 

        nodes.add(root);
        traversal(root.left, nodes);
        traversal(root.right, nodes);
        root.left=null; root.right=null; 
    }


}


//Second approach --- disconnect left , right subtrees, flatten left, right in recursion,,, connect in a way of linkedlist (right skewed tree)

class Solution {
    public void flatten(TreeNode root) {
        if(root==null) return; 

        TreeNode lst=root.left; 
        TreeNode rst=root.right; 
        root.left=null; root.right=null; 

        flatten(lst);
        flatten(rst);

        TreeNode last= root; 
        if(lst!=null){
            root.right=lst; 
            while(last.right!=null) last=last.right; 
        }
        
        last.right=rst;
    }
}




//Third Approach --- using Modified Morris traversal


class Solution {
    public void flatten(TreeNode root) {
        TreeNode curr=root; 
        while(curr!=null){
            if(curr.left!=null){
                //(((find pred, link, unlink)))

                //1) find pred
                //2) here, we will link pred to curr.right && curr.right to curr.left; (fake connection)   
                //3) unlink --> curr.left=null ,,, curr=curr.right

            
                TreeNode pred=curr.left;
                while(pred.right!=null && pred.right!=curr) pred=pred.right; 
                
                pred.right=curr.right; 
                curr.right=curr.left; 
                curr.left=null;
                curr=curr.right; 

            }
            else{     //if pred not being able to found --- c=c.right
                curr=curr.right;
            }
        } 
         
    }

}
