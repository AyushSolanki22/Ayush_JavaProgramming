class MedianFinder {  //we are creating some data structure
    PriorityQueue<Integer> minHeap=new PriorityQueue<>();
    PriorityQueue<Integer> maxHeap=new PriorityQueue<>(Collections.reverseOrder());
   
    public MedianFinder() {
        
    }
    
    //smaller n/2 elements - maxHeap,,,, larger n/2 elements - minHeap

    //if element being added is more than max of smaller elements, then it is added to larger elements,, and vice versa
    //after adding , remember -- if difference of size >1 ,, peek of more ele gets removed and added to another
    public void addNum(int num) {
        if(maxHeap.size()==0) {   //base
            maxHeap.add(num);  
            return;
        }

        if(num>maxHeap.peek()) minHeap.add(num); 
        else maxHeap.add(num);

        //after adding handling ((if size diff>1))
        int maxSize=maxHeap.size();
        int minSize=minHeap.size();
        if(Math.abs(minSize-maxSize)>1) {
            if(minSize>maxSize) maxHeap.add(minHeap.remove());
            else minHeap.add(maxHeap.remove());
        }

    }
    
    //for finding median simultaneously, having the array divided into 2 parts, smaller elements, larger elements ,, maxHeap for smaller elements, minHeap for larger elements,,,,,, 
    //in case of even elements, so for median, max element from smaller elements, min elemenet from larger elements to get mean 
    //in case of odd elements, -- both heaps size are unequal, --- one with more elements' peek will be the median
    public double findMedian() {
        int maxSize=maxHeap.size();
        int minSize=minHeap.size();
        if(maxSize > minSize) return maxHeap.peek();
        else if(minSize > maxSize) return minHeap.peek();
        else{
            return (minHeap.peek()+maxHeap.peek() )/2.0;
        }
    }
}

//

//

// The median depends on the middle element(s) of sorted data.
// We don't need the complete sorted order.
