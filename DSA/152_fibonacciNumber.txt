//One approach -- using array to store fibo results, (dp )
//Memorisation / top down DP / Recursive DP

//T.C= 0(n), A.S = 0(n)
class Solution {
    static int dp[];
    public int fib(int n) {
        dp=new int[n+1];  
        Arrays.fill(dp, 0); 
        return fibo(n);
    }

    static int fibo(int n){
        if (n==0 || n==1) {
                    return  n;
        }        
        if(dp[n]!=0){
            return dp[n];   //returning stored fibo(n) 
        }
        int fibo=fibo(n-1)+fibo(n-2);
        dp[n]=fibo;    //storing fibo(n)
        return fibo;
    }
}


//Second approach -- using Tabulation method/iterative dp

class Solution {
    public int fib(int n) {
        if(n<=1) return n;
        
        int dp[]=new int[n+1];
        dp[0]=0; 
        dp[1]=1; 
        
        for(int i=2; i<=n; ++i){
            dp[i]=dp[i-1]+dp[i-2];
        }
        
        return dp[n];
    }
    
}



//Third Approach - Space Optimised Tabulation

class Solution {
    public int fib(int n) {
        if(n<=1) return n;
        
        int dp[]=new int[3];
        dp[0]=0; 
        dp[1]=1; 
        
        for(int i=2; i<=n; ++i){
            dp[2]=dp[1]+dp[0];
            dp[0]=dp[1];
            dp[1]=dp[2];
        }
        
        return dp[2];
    }
}
